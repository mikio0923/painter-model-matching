## 🤖 自動化スクリプト案

### HTTP スモークテスト（全ページ 200/302 確認）

```bash
#!/bin/bash
BASE="http://220.158.26.223"

# 未ログインで開けるべきページ
PAGES=(
  "/" "/about" "/faq" "/guideline" "/privacy" "/terms"
  "/guide/model" "/guide/painter"
  "/models" "/jobs" "/information"
  "/contact" "/login-register"
  "/register/email"
  "/password/reset"
  "/admin/login"
)

PASS=0
FAIL=0
for p in "${PAGES[@]}"; do
    CODE=$(curl -s -o /dev/null -w "%{http_code}" "${BASE}${p}")
    if [ "$CODE" = "200" ] || [ "$CODE" = "302" ]; then
        echo "✓ $CODE  $p"
        PASS=$((PASS+1))
    else
        echo "✗ $CODE  $p"
        FAIL=$((FAIL+1))
    fi
done

echo ""
echo "結果: $PASS 成功 / $FAIL 失敗"
```
